import pandas as pd
df = pd.read_csv("https://sozerong.github.io/home_in_seoul-data/v2026.11.16/dong_profile.csv", dtype={"dong_id": str, "mois_dong_cd": str, "sgg_cd": str})

# 최신 버전 찾기
import json, urllib.request
latest = json.load(urllib.request.urlopen("https://sozerong.github.io/home_in_seoul-data/latest.json"))["version"]
